# ModalNavigationDrawer（模态抽屉）

```
@Composable
fun SimpleDrawerDemo() {
    // 1. 声明抽屉的状态（默认是关闭的 Closed）
    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
    // 2. 声明一个协程作用域（Compose中触发动画/滑出必须在协程里运行）
    val scope = rememberCoroutineScope()

    // 3. 官方抽屉组件（相当于 Vue 里的抽屉外层组件）
    ModalNavigationDrawer(
        drawerState = drawerState, // 绑定状态
        drawerContent = {
            // 这里是左侧滑出来的导航栏内容（相当于 CSS 伸缩的那部分面板）
            ModalDrawerSheet(modifier = Modifier.width(250.dp)) {
                Text("我是侧边栏内容", modifier = Modifier.padding(16.dp))
            }
        }
    ) {
        // 4. 这里是屏幕的主体内容
        Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {
            // 你的圆形头像，加上 clickable 就能点了
            Box(
                modifier = Modifier
                    .size(45.dp)
                    .clip(CircleShape)
                    .background(Color.Blue)
                    .clickable {
                        // 5. 点击头像，在协程里调用 open() 方法，侧边栏就会自己滑出来
                        scope.launch { drawerState.open() }
                    }
            )
            Text("点击蓝色圆形头像打开抽屉")
        }
    }
}
```
